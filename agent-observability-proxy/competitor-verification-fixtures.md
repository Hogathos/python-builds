```python
"""Throwaway local MCP upstream for verification runs. Metadata-only log: never records args or results."""
import json, os, sys, time
from pathlib import Path
from mcp.server.fastmcp import FastMCP

LOG = os.environ.get("UPSTREAM_LOG")
WATCH = os.environ.get("WATCH_LOG")  # a product log file; we only count its lines when a call arrives
NOTES = Path(os.environ.get("NOTES_DIR", "/home/claude/verify/notes"))
NOTES.mkdir(parents=True, exist_ok=True)

def _lines(p):
    try:
        if str(p).endswith(".db"):
            import sqlite3
            c = sqlite3.connect(f"file:{p}?mode=ro", uri=True, timeout=2)
            try:
                return c.execute("SELECT COUNT(*) FROM audit_log").fetchone()[0]
            finally:
                c.close()
        with open(p, "rb") as f:
            return sum(1 for _ in f)
    except Exception:
        return None

def meta(ev, **kw):
    if not LOG:
        return
    rec = {"ev": ev, "pid": os.getpid(), "t_ns": time.time_ns(), **kw}
    with open(LOG, "a") as f:
        f.write(json.dumps(rec) + "\n")

meta("spawn")
SO = os.environ.get("STRUCTURED_OUTPUT", "1") == "1"
mcp = FastMCP("iris-test-upstream")

def _seen(tool):
    meta("call", tool=tool, watch_lines=_lines(WATCH) if WATCH else None)

@mcp.tool(structured_output=SO)
def ping() -> str:
    """Harmless read-only ping."""
    _seen("ping"); return "pong"

@mcp.tool(structured_output=SO)
def read_note(name: str) -> str:
    """Read a note from the scratch dir."""
    _seen("read_note")
    p = NOTES / name
    return p.read_text() if p.exists() else "(missing)"

@mcp.tool(structured_output=SO)
def write_note(name: str, text: str) -> str:
    """Write a note into the scratch dir (mutating)."""
    _seen("write_note")
    (NOTES / name).write_text(text)
    return "written"

@mcp.tool(structured_output=SO)
def delete_note(name: str) -> str:
    """Delete a note from the scratch dir (mutating)."""
    _seen("delete_note")
    (NOTES / name).unlink(missing_ok=True)
    return "deleted"

@mcp.tool(structured_output=SO)
def write_then_die(name: str, text: str) -> str:
    """Mutating: commits the effect, then the process dies before replying (crash-after-commit)."""
    _seen("write_then_die")
    (NOTES / name).write_text(text)
    os._exit(1)

if __name__ == "__main__":
    mcp.run()
```

```python
"""Minimal scripted MCP stdio client (stand-in for Claude Desktop). Reads a claude_desktop_config-style file,
launches ONE named entry, runs steps, prints METADATA ONLY per step."""
import asyncio, hashlib, json, sys, time
from mcp import ClientSession, StdioServerParameters
from mcp.client.stdio import stdio_client

async def main(cfg_path, entry, steps_path, errlog_path):
    cfg = json.load(open(cfg_path))["mcpServers"][entry]
    steps = json.load(open(steps_path))
    params = StdioServerParameters(command=cfg["command"], args=cfg.get("args", []), env=cfg.get("env"), cwd=cfg.get("cwd"))
    with open(errlog_path, "a") as errlog:
        async with stdio_client(params, errlog=errlog) as (r, w):
            async with ClientSession(r, w) as s:
                init = await asyncio.wait_for(s.initialize(), 40)
                print(json.dumps({"step": "initialize", "server": init.serverInfo.name, "server_ver": init.serverInfo.version, "protocol": init.protocolVersion}))
                for i, st in enumerate(steps):
                    t0 = time.time()
                    if st.get("list"):
                        res = await asyncio.wait_for(s.list_tools(), 40)
                        print(json.dumps({"step": i, "op": "tools/list", "tools": sorted(t.name for t in res.tools)}))
                        continue
                    try:
                        res = await asyncio.wait_for(s.call_tool(st["tool"], st.get("args", {})), 40)
                        txt = "".join(getattr(c, "text", "") for c in res.content)
                        print(json.dumps({"step": i, "op": "tools/call", "tool": st["tool"], "isError": bool(res.isError),
                              "n_content": len(res.content), "text_len": len(txt), "sha8": hashlib.sha256(txt.encode()).hexdigest()[:8],
                              "head": txt.split("\n")[0][:80], "ms": int((time.time() - t0) * 1000)}))
                    except Exception as e:
                        print(json.dumps({"step": i, "op": "tools/call", "tool": st["tool"], "exception": type(e).__name__, "msg": str(e)[:120]}))
asyncio.run(main(*sys.argv[1:5]))
```
